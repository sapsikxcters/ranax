local function safeDestroy(parent, name)
    if not parent then return end
    local obj = parent:FindFirstChild(name)
    if obj then obj:Destroy() end
end

task.spawn(function()
    _G.LuauPro_Launcher.Prepare()
    _G.LuauPro_Launcher.Play()

    for i = 1, 100, 5 do
        _G.LuauPro_Launcher.SetProgress(i)
        task.wait(0.05)
    end

    _G.LuauPro_Launcher.SetProgress(100, "Ready — choose interface")
    task.wait(0.4)

    _G.LuauPro_Launcher.AskInterface(function(choice)
        if choice == "mobile" then
            _G.LuauPro_Launcher.SetProgress(100, "Mobile layout selected")
        else
            _G.LuauPro_Launcher.SetProgress(100, "PC layout selected")
        end
        task.wait(0.8)
        ScreenGui3:Destroy()
    end)
end)
